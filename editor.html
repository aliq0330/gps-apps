function selamVer(isim) {
    const mesaj = "Merhaba, " + isim + "!";
    console.log(mesaj);
    
    // Döngü örneği
    for(let i = 0; i < 3; i++) {
        console.log(i + ". tekrar");
    }
    
    return mesaj;
}

selamVer("Dünya");
